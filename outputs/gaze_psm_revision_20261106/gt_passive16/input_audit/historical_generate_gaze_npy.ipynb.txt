    "# Math function (Angle-axis rotation + projection)\n",
    "def angle_axis_to_R(ax: float, ay: float, az: float) -> np.ndarray:\n",
    "    \"\"\"\n",
    "    Convert angle-axis vector (ax,ay,az) to 3x3 rotation matrix.\n",
    "    Here (ax,ay,az) encodes axis * angle (Rodrigues vector).\n",
    "    \"\"\"\n",
    "    r = np.array([ax, ay, az], dtype=float)\n",
    "    theta = np.linalg.norm(r)\n",
    "    if theta < 1e-12:\n",
    "        return np.eye(3, dtype=float)\n",
    "\n",
    "    k = r / theta\n",
    "    kx, ky, kz = k\n",
    "    K = np.array([\n",
    "        [0,  -kz, ky],\n",
    "        [kz,  0, -kx],\n",
    "        [-ky, kx,  0]\n",
    "    ], dtype=float)\n",
    "\n",
    "    R = np.eye(3, dtype=float) + math.sin(theta) * K + (1 - math.cos(theta)) * (K @ K)\n",
    "    return R\n",
    "\n",
    "\n",
    "def project_gaze_to_pixel(gx, gy, gz, view_center_x, win_w, win_h,\n",
    "                          d, ox, oy, ax, ay, az, use_pose=True):\n",
    "    cx, cy = int(view_center_x), int(win_h / 2)\n",
    "    if gz == 0.0:\n",
    "        return cx, cy\n",
    "\n",
    "    if not use_pose:\n",
    "        z_ratio = d / gz\n",
    "        px = int(gx * z_ratio + cx) + int(ox)\n",
    "        py = int(win_h - (gy * z_ratio + cy)) + int(oy)\n",
    "        return px, py\n",
    "\n",
    "    R = angle_axis_to_R(ax, ay, az)\n",
    "    u = R @ np.array([1, 0, 0], float)\n",
    "    w = -(R @ np.array([0, 1, 0], float))  # pixel Y down\n",
    "    n = R @ np.array([0, 0, 1], float)\n",
    "\n",
    "    v = np.array([gx, gy, gz], float)\n",
    "    denom = float(n @ v)\n",
    "    if abs(denom) < 1e-8:\n",
    "        return cx, cy\n",
    "\n",
    "    t = d / denom\n",
    "    X = t * v\n",
    "    Xu = float(u @ X)\n",
    "    Xw = float(w @ X)\n",
    "\n",
    "    px = int(Xu + cx) + int(ox)\n",
    "    py = int(Xw + cy) + int(oy)\n",
    "    return px, py\n"

    "# Alignment core (nearest match with optional max_dt)\n",
    "def nearest_align_frames_to_gaze(\n",
    "    t_frames_ns: np.ndarray,\n",
    "    t_gaze_ns: np.ndarray,\n",
    "    gaze_xy: np.ndarray,     # shape [Ng,2]\n",
    "    gaze_valid: np.ndarray,  # shape [Ng]\n",
    "    max_dt_ms: Optional[float] = None\n",
    ") -> Tuple[np.ndarray, np.ndarray, np.ndarray]:\n",
    "    \"\"\"\n",
    "    For each t_frame, pick nearest t_gaze.\n",
    "    Returns:\n",
    "      aligned_xy: [Nf,2] float32 (may contain NaN if max_dt_ms is set and exceeded)\n",
    "      aligned_valid: [Nf] uint8/int\n",
    "      dt_abs_ns: [Nf] int64 absolute dt to nearest gaze (for QC)\n",
    "    \"\"\"\n",
    "    if len(t_gaze_ns) == 0 or len(t_frames_ns) == 0:\n",
    "        aligned_xy = np.full((len(t_frames_ns), 2), np.nan, dtype=np.float32)\n",
    "        aligned_valid = np.zeros((len(t_frames_ns),), dtype=np.uint8)\n",
    "        dt_abs_ns = np.full((len(t_frames_ns),), np.iinfo(np.int64).max, dtype=np.int64)\n",
    "        return aligned_xy, aligned_valid, dt_abs_ns\n",
    "\n",
    "    # Ensure gaze timestamps sorted\n",
    "    order = np.argsort(t_gaze_ns)\n",
    "    t_gaze_ns = t_gaze_ns[order]\n",
    "    gaze_xy = gaze_xy[order]\n",
    "    gaze_valid = gaze_valid[order]\n",
    "\n",
    "    # searchsorted\n",
    "    idx = np.searchsorted(t_gaze_ns, t_frames_ns, side=\"left\")\n",
    "\n",
    "    # candidates: idx-1 and idx\n",
    "    idx0 = np.clip(idx - 1, 0, len(t_gaze_ns) - 1)\n",
    "    idx1 = np.clip(idx,     0, len(t_gaze_ns) - 1)\n",
    "\n",
    "    dt0 = np.abs(t_frames_ns - t_gaze_ns[idx0])\n",
    "    dt1 = np.abs(t_frames_ns - t_gaze_ns[idx1])\n",
    "\n",
    "    choose1 = dt1 < dt0\n",
    "    best_idx = np.where(choose1, idx1, idx0)\n",
    "    dt_abs = np.where(choose1, dt1, dt0).astype(np.int64)\n",
    "\n",
    "    aligned_xy = gaze_xy[best_idx].astype(np.float32, copy=False)\n",
    "    aligned_valid = gaze_valid[best_idx]\n",
    "\n",
    "    if max_dt_ms is not None:\n",
    "        max_dt_ns = int(max_dt_ms * 1e6)  # ms -> ns\n",
    "        mask = dt_abs > max_dt_ns\n",
    "        if np.any(mask):\n",
    "            aligned_xy = aligned_xy.copy()\n",
    "            aligned_xy[mask, :] = np.nan  # keep valid as-is (or set to 0 if you prefer)\n",
    "    return aligned_xy, aligned_valid, dt_abs\n"
